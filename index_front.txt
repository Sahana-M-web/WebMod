           placeholder="Enter food name">

    <input type="number"
           ng-model="foodPrice"
           placeholder="Enter price">

    <button class="add" ng-click="saveFood()">
        {{ editIndex === -1 ? 'Add Food' : 'Update Food' }}
    </button>

    <br>

    <input type="text"
           ng-model="searchText"
           placeholder="Search food">

            <th>Actions</th>
        </tr>

        <tr class="food-row"

            <td>{{ food.name | capitalize }}</td>
            <td>{{ food.price | currency:'₹' }}</td>

            <td>
                <button class="edit" ng-click="editFood(food)">
                    Edit
                </button>

                <button class="delete" ng-click="deleteFood(food)">

<script>
var app = angular.module("foodApp", ["ngAnimate"]);

// Custom filter
app.filter("capitalize", function() {
    return function(text) {
        if (!text) return "";
        return text.charAt(0).toUpperCase() + text.slice(1);
    };
});

// Factory
app.factory("FoodFactory", function() {
    return {
        createFood: function(name, price) {
            return {
                name: name,
                price: Number(price)
            };
        }
    };
});

// Service
app.service("FoodService", function(FoodFactory) {
    this.getFoods = function() {
        return [
            FoodFactory.createFood("Pizza", 200),
            FoodFactory.createFood("Burger", 100),
            FoodFactory.createFood("Dosa", 60)
        ];
    };
});

// Controller with dependency injection
app.controller("foodController", function(
    $scope, FoodService, FoodFactory
) {
    $scope.foods = FoodService.getFoods();
    $scope.editIndex = -1;

    // Create and update
    $scope.saveFood = function() {
        if (!$scope.foodName ||
            $scope.foodPrice === undefined ||
            $scope.foodPrice === null ||
            $scope.foodPrice === "" ||
            Number($scope.foodPrice) < 0) {
            alert("Enter a valid food name and price");
            return;
        }

        if ($scope.editIndex === -1) {
            $scope.foods.push(
                FoodFactory.createFood(
                    $scope.foodName,
                    $scope.foodPrice
                )
            );
        } else {
            $scope.foods[$scope.editIndex].name =
                $scope.foodName;

            $scope.foods[$scope.editIndex].price =
                Number($scope.foodPrice);
        }

        $scope.foodName = "";
        $scope.foodPrice = "";
        $scope.editIndex = -1;
    };

    // Edit
    $scope.editFood = function(food) {
        $scope.foodName = food.name;
        $scope.foodPrice = food.price;
        $scope.editIndex = $scope.foods.indexOf(food);
    };

    // Delete
    $scope.deleteFood = function(food) {
        if (confirm("Delete this food item?")) {
            $scope.foods.splice(
                $scope.foods.indexOf(food), 1
            );

            if ($scope.editIndex === $scope.foods.indexOf(food)) {
                $scope.editIndex = -1;
                $scope.foodName = "";
                $scope.foodPrice = "";
            }
        }
    };
});
</script>
