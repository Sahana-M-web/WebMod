transform.py


from lxml import etree

# Read the XML file
xml = etree.parse("student.xml")

# Read the XSLT file
xslt = etree.parse("student.xsl")

# Create the transformation
transform = etree.XSLT(xslt)

# Transform XML into HTML
result = transform(xml)

# Save the HTML output
with open("student.html", "wb")as file:
    file.write(etree.toString(result,
method="html",
pretty_print=True,
encoding="UTF-8"))
