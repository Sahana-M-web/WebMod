index.html

<!DOCTYPE html>
<html ng-app="bookApp">

<head>
    <title>Book Management</title>

    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
</head>

<body ng-controller="bookController">

    <h2>Book Management System</h2>

    <input
        type="text"
        id="bookName"
        ng-model="bookName"
        placeholder="Enter book name"
    >

    <button id="addBtn" ng-click="addBook()">
        Add Book
    </button>

    <p id="result">{{message}}</p>

    <script>
        var app = angular.module("bookApp", []);

        app.controller("bookController", function($scope) {

            $scope.addBook = function() {

                if ($scope.bookName && $scope.bookName.trim()) {

                    $scope.message = "Book added: " + $scope.bookName;

                } else {

                    $scope.message = "Please enter a book name";

                }
            };

        });
    </script>

</body>
</html>


create test.js



const { Builder, By, until } = require("selenium-webdriver");

async function testBookApp() {

    const driver = await new Builder()
        .forBrowser("chrome")
        .build();

    let failed = false;

    try {

        await driver.get("http://localhost:3000");

        // TEST 1: valid input
        await driver.findElement(By.id("bookName"))
            .sendKeys("Java Programming");

        await driver.findElement(By.id("addBtn"))
            .click();

        const result = await driver.findElement(By.id("result"));

        await driver.wait(
            until.elementTextIs(result, "Book added: Java Programming"),
            5000
        );

        console.log("TEST 1 PASSED: book added message is correct");

        // TEST 2: empty input shows an error message
        await driver.findElement(By.id("bookName")).clear();

        await driver.findElement(By.id("addBtn")).click();

        await driver.wait(
            until.elementTextIs(result, "Please enter a book name"),
            5000
        );

        console.log("TEST 2 PASSED: empty input validation works");

    } catch (error) {

        failed = true;
        console.log("TEST FAILED:", error.message);

    } finally {

        await driver.quit();
        process.exit(failed ? 1 : 0);

    }
}

testBookApp();
